   107→   107→function uid(): string {
   108→   108→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
   109→   109→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`

   112→   112→function truncate(s: string, n: number): string {
   113→   113→  const t = s.trim().replace(/\s+/g, ' ')
   114→   114→  return t.length <= n ? t : t.slice(0, n - 1) + '…'
   115→   115→}

   117→   117→/** Bloc texte des pièces jointes, injecté dans le prompt du modèle */
   118→   118→function attachmentBlock(m: Message, maxPerFile = 12000): string {
   119→   119→  if (!m.attachments?.length) return ''
   120→   120→  const parts = m.attachments.map((a) =>
   121→   121→    a.text
   122→   122→      ? `--- Fichier joint : ${a.name} ---\n${a.text.slice(0, maxPerFile)}`
   123→   123→      : `[Pièce jointe : ${a.name} (${a.type || 'fichier'}, ${(a.size / 1024).toFixed(0)} Ko)]`,
   124→   124→  )
   125→   125→  return '\n\n[Fichiers joints par l’utilisateur]\n' + parts.join('\n\n')
   126→   126→}
   127→   127→
   128→   128→function attachmentSummary(attachments?: Attachment[]): string {
   129→   129→  if (!attachments?.length) return ''
   130→   130→  const names = attachments.map((a) => a.name).join(', ')
   131→   131→  const texts = attachments
   132→   132→    .filter((a) => a.text)
   133→   133→    .map((a) => `--- ${a.name} ---\n${a.text!.slice(0, 2000)}`)
   134→   134→    .join('\n\n')
   135→   135→  return `\n\n(En tenant compte des fichiers joints : ${names})${texts ? '\n\n' + texts : ''}`
   136→   136→}

   144→   144→      function updateMessage(sessionId: string, msgId: string, patch: Partial<Message>) {
   145→   145→        updateSession(sessionId, (s) => ({
   146→   146→          ...s,
   147→   147→          updatedAt: Date.now(),
   148→   148→          messages: s.messages.map((m) => (m.id === msgId ? { ...m, ...patch } : m)),
   149→   149→        }))

   156→   156→      function revealStream(sessionId: string, msgId: string, full: string): Promise<string> {
   157→   157→        return new Promise((resolve) => {
   158→   158→          let revealed = 0
   159→   159→          const chunk = Math.max(3, Math.ceil(full.length / 160))
   160→   160→          const timer = setInterval(() => {
   161→   161→            revealed = Math.min(full.length, revealed + chunk)
   162→   162→            updateMessage(sessionId, msgId, { content: full.slice(0, revealed) })
   163→   163→            if (revealed >= full.length) {
   164→   164→              clearInterval(timer)
   165→   165→              streams.delete(msgId)
   166→   166→              updateMessage(sessionId, msgId, { content: full, pending: false })
   167→   167→              resolve(full)
   168→   168→            }
   169→   169→          }, 16)
   170→   170→          streams.set(msgId, {
   171→   171→            timer,
   172→   172→            sessionId,
   173→   173→            resolve: (content) => {
   174→   174→              updateMessage(sessionId, msgId, { content, pending: false, stopped: true })
   175→   175→              resolve(content)
   176→   176→            },
   177→   177→          })
   178→   178→        })
   179→   179→      }
   180→   180→
   181→   181→      function later(sessionId: string, ms: number, fn: () => void) {
   182→   182→        const t = setTimeout(() => {
   183→   183→          const set_ = sessionTimers.get(sessionId)
   184→   184→          if (set_) set_.delete(t)
   185→   185→          fn()
   186→   186→        }, ms)
   187→   187→        if (!sessionTimers.has(sessionId)) sessionTimers.set(sessionId, new Set())
   188→   188→        sessionTimers.get(sessionId)!.add(t)
   189→   189→      }

   193→   193→      async function chatFlow(
   194→   194→        sessionId: string,
   195→   195→        history: Message[],
   196→   196→        override?: { provider?: string; hfModel?: string },
   197→   197→      ) {
   198→   198→        const session = get().sessions.find((s) => s.id === sessionId)
   199→   199→        if (!session) return
   200→   200→        const providerLabel =
   201→   201→          override?.provider === 'hf'
   202→   202→            ? `hf:${override.hfModel}`
   203→   203→            : (override?.provider ?? session.models.text)
   204→   204→        const assistantId = uid()
   205→   205→        addMessage(sessionId, {
   206→   206→          id: assistantId,
   207→   207→          role: 'assistant',
   208→   208→          content: '',
   209→   209→          createdAt: Date.now(),
   210→   210→          pending: true,
   211→   211→          flow: 'chat',
   212→   212→          provider: providerLabel,
   213→   213→        })
   214→   214→        try {
   215→   215→          const turns = history
   216→   216→            .filter((m) => !m.pending && !m.error && m.content.trim())
   217→   217→            .slice(-16)
   218→   218→            .map((m) => ({ role: m.role, content: m.content + attachmentBlock(m) }) as { role: 'user' | 'assistant'; content: string })
   219→   219→          let system = LUMO_SYSTEM
   220→   220→          if (session.mode === 'project') system = PROJECT_SYSTEM
   221→   221→          if (session.mode === 'book' && session.brief) {
   222→   222→            const bibleArtifact = session.messages
   223→   223→              .flatMap((m) => m.artifacts ?? [])
   224→   224→              .find((a) => a.kind === 'bible')
   225→   225→            system = bookSystem(
   226→   226→              bibleArtifact?.text ??
   227→   227→                'Brief : ' + JSON.stringify(session.brief, null, 2),
   228→   228→            )
   229→   229→          }
   230→   230→          const useOverride: ChatOverride | undefined =
   231→   231→            override?.provider && override.provider !== 'lumo-core'
   232→   232→              ? { provider: override.provider as ChatOverride['provider'], hfModel: override.hfModel }
   233→   233→              : undefined
   234→   234→          const full = await apiChat(turns, system, useOverride)
   235→   235→          const stillPending = get().sessions
   236→   236→            .find((s) => s.id === sessionId)
   237→   237→            ?.messages.find((m) => m.id === assistantId)?.pending
   238→   238→          if (stillPending) await revealStream(sessionId, assistantId, full)
   239→   239→          else updateMessage(sessionId, assistantId, { content: full, pending: false })
   240→   240→        } catch (e) {
   241→   241→          const err = e as Error & { fallback?: string[] }
   242→   242→          updateMessage(sessionId, assistantId, {
   243→   243→            pending: false,
   244→   244→            error: err.message,
   245→   245→            content: '',
   246→   246→            fallback: err.fallback ?? ['mistral', 'agnes', 'hf'],
   247→   247→          })
   248→   248→        }
   249→   249→      }

   251→   251→      async function imageFlow(sessionId: string, prompt: string, opts: SendOptions) {
   252→   252→        const session = get().sessions.find((s) => s.id === sessionId)
   253→   253→        if (!session) return
   254→   254→        const count = Math.max(1, Math.min(4, opts.variations ?? 1))
   255→   255→        const size = opts.imageSize ?? '1024x1024'
   256→   256→        const provider = session.models.image
   257→   257→        const fullPrompt = prompt + attachmentSummary(opts.attachments)
   258→   258→        const assistantId = uid()
   259→   259→        addMessage(sessionId, {
   260→   260→          id: assistantId,
   261→   261→          role: 'assistant',
   262→   262→          content: '',
   263→   263→          createdAt: Date.now(),
   264→   264→          pending: true,
   265→   265→          flow: 'image',
   266→   266→          provider,
   267→   267→        })
   268→   268→        const results = await Promise.allSettled(
   269→   269→          Array.from({ length: count }, (_, i) =>
   270→   270→            i === 0
   271→   271→              ? apiImage(fullPrompt, size)
   272→   272→              : new Promise<{ base64: string; bytes: number }>((resolve, reject) =>
   273→   273→                  setTimeout(() => apiImage(fullPrompt, size).then(resolve, reject), i * 3000),
   274→   274→                ),
   275→   275→          ),
   276→   276→        )
   277→   277→        const artifacts: Artifact[] = []
   278→   278→        let ok = 0
   279→   279→        for (let i = 0; i < results.length; i++) {
   280→   280→          const r = results[i]
   281→   281→          if (r.status === 'fulfilled') {
   282→   282→            ok++
   283→   283→            const a: Artifact = {
   284→   284→              id: uid(),
   285→   285→              kind: 'image',
   286→   286→              name: `image_${String(artifacts.length + 1).padStart(3, '0')}.png`,
   287→   287→              mimeType: 'image/png',
   288→   288→              createdAt: Date.now(),
   289→   289→              provider,
   290→   290→              prompt: fullPrompt,
   291→   291→              data: r.value.base64,
   292→   292→              bytes: r.value.bytes,
   293→   293→              meta: { size },
   294→   294→            }
   295→   295→            void putArtifactData(a.id, a.data)
   296→   296→            artifacts.push(a)
   297→   297→          }
   298→   298→        }
   299→   299→        const failed = count - ok
   300→   300→        const content =
   301→   301→          ok === 0
   302→   302→            ? `La génération d’image a échoué pour « ${truncate(prompt, 60)} ». Réessayez dans un instant.`
   303→   303→            : `Voici ${ok > 1 ? `les ${ok} variations` : 'votre visuel'} pour « ${truncate(prompt, 60)} ».${
   304→   304→                failed ? ` (${failed} tentative(s) échouée(s))` : ''
   305→   305→              }`
   306→   306→        updateMessage(sessionId, assistantId, {
   307→   307→          content,
   308→   308→          pending: false,
   309→   309→          artifacts,
   310→   310→          error: ok === 0 ? 'Génération impossible' : undefined,
   311→   311→        })
   312→   312→      }
