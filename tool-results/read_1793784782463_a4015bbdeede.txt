   138→function ElView({ el, palette, selected, onSelect }: { el: DesignEl; palette: DesignDoc['palette']; selected: boolean; onSelect: () => void }) {
   139→  const p = el.props
   140→  const ring = selected ? 'outline-2 outline-offset-2 outline-primary' : 'hover:outline-2 hover:outline-offset-2 hover:outline-primary/40'
   141→  const common = `cursor-pointer transition-all ${ring}`
   142→  switch (el.type) {
   143→    case 'navbar':
   144→      return (
   145→        <header onClick={onSelect} className={`flex flex-wrap items-center justify-between gap-2 rounded-2xl bg-white p-3 shadow-sm ${common}`} role="presentation">
   146→          <span className="font-bold" style={{ color: palette.primary }}>{p.text || 'Mon app'}</span>
   147→          <span className="flex flex-wrap gap-3 text-xs opacity-75">
   148→            {(p.links ?? '').split(/[,\n]/).map((l) => l.trim()).filter(Boolean).map((l, i) => <span key={i}>{l}</span>)}
   149→          </span>
   150→        </header>
   151→      )
   152→    case 'hero':
   153→      return (
   154→        <section onClick={onSelect} className={`flex flex-col items-center gap-2 rounded-2xl bg-white p-6 text-center shadow-sm sm:p-9 ${common}`} role="presentation">
   155→          <h3 className="text-xl font-bold sm:text-2xl" style={{ color: palette.text }}>{p.title}</h3>
   156→          <p className="max-w-md text-xs opacity-70 sm:text-sm">{p.subtitle}</p>
   157→          <span className="rounded-full px-5 py-2 text-xs font-semibold text-white" style={{ background: palette.primary }}>{p.cta}</span>
   158→        </section>
   159→      )
   160→    case 'text':
   161→      return (
   162→        <p
   163→          onClick={onSelect}
   164→          className={`px-1 opacity-90 ${common} ${p.size === 'lg' ? 'text-base font-medium sm:text-lg' : p.size === 'sm' ? 'text-[11px]' : 'text-xs sm:text-sm'}`}
   165→          role="presentation"
   166→        >
   167→          {p.text}
   168→        </p>
   169→      )
   170→    case 'button':
   171→      return (
   172→        <p className="px-1" role="presentation">
   173→          <button
   174→            onClick={(e) => {
   175→              e.stopPropagation()
   176→              onSelect()
   177→            }}
   178→            className={`rounded-full px-4 py-2 text-xs font-semibold transition-colors sm:text-sm ${common} ${
   179→              p.variant === 'outline' ? 'border-2 bg-transparent' : 'text-white'
   180→            }`}
   181→            style={p.variant === 'outline' ? { borderColor: palette.primary, color: palette.primary } : { background: palette.primary }}
   182→          >
   183→            {p.text}
   184→          </button>
   185→        </p>
   186→      )
   187→    case 'input':
   188→      return (
   189→        <label onClick={onSelect} className={`flex cursor-pointer flex-col gap-1 rounded-xl bg-white/60 px-1 py-1 text-[11px] ${common}`} role="presentation">
   190→          <span className="opacity-70">{p.label}</span>
   191→          <span className="rounded-lg border border-black/15 bg-white px-3 py-2 text-muted-foreground">{p.placeholder}</span>
   192→        </label>
   193→      )
   194→    case 'image':
   195→      return (
   196→        <div
   197→          onClick={onSelect}
   198→          className={`grid min-h-24 place-items-center rounded-2xl text-xs opacity-80 sm:min-h-32 ${common}`}
   199→          style={{ background: `color-mix(in srgb, ${palette.primary} 16%, white)` }}
   200→          role="presentation"
   201→        >
   202→          🖼 {p.alt || 'Image'}
   203→        </div>
   204→      )
   205→    case 'list':
   206→      return (
   207→        <ul onClick={onSelect} className={`list-disc space-y-1 rounded-2xl bg-white p-4 pl-8 text-xs shadow-sm sm:text-sm ${common}`} role="presentation">
   208→          {(p.items ?? '').split(/\n/).map((i) => i.trim()).filter(Boolean).map((i, k) => <li key={k}>{i}</li>)}
   209→        </ul>
   210→      )
   211→    case 'card':
   212→      return (
   213→        <div onClick={onSelect} className={`rounded-2xl bg-white p-4 shadow-sm ${common}`} role="presentation">
   214→          <p className="mb-1 text-sm font-semibold" style={{ color: palette.primary }}>{p.title}</p>
   215→          <p className="text-xs opacity-75">{p.text}</p>
   216→        </div>
   217→      )
   218→    case 'badge':
   219→      return (
   220→        <p className="px-1" role="presentation">
   221→          <span onClick={onSelect} className={`inline-block cursor-pointer rounded-full px-3 py-1 text-[10px] font-semibold sm:text-xs ${common}`} style={{ background: `color-mix(in srgb, ${palette.primary} 18%, white)`, color: palette.primary }}>
   222→            {p.text}
   223→          </span>
   224→        </p>
   225→      )
   226→    case 'divider':
   227→      return <hr onClick={onSelect} className={`cursor-pointer border-black/10 ${common}`} role="presentation" />
   228→    case 'footer':
   229→      return (
   230→        <p onClick={onSelect} className={`cursor-pointer pt-2 text-center text-[11px] opacity-60 ${common}`} role="presentation">
   231→          {p.text}
   232→        </p>
   233→      )
   234→    default:
   235→      return null
   236→  }
   237→}

   334→
   335→function PropsPanel({
   336→  doc,
   337→  setDoc,
   338→  activeScreenId,
   339→  selected,
   340→  setSelected,
   341→}: {
   342→  doc: DesignDoc
   343→  setDoc: (fn: (d: DesignDoc) => DesignDoc) => void
   344→  activeScreenId: string
   345→  selected: DesignEl | null
   346→  setSelected: (id: string | null) => void
   347→}) {
   348→  const screen = doc.screens.find((s) => s.id === activeScreenId)
   349→  function patchProp(key: string, value: string) {
   350→    if (!selected || !screen) return
   351→    setDoc((d) => ({
   352→      ...d,
   353→      screens: d.screens.map((s) =>
   354→        s.id === screen.id
   355→          ? {
   356→              ...s,
   357→              elements: s.elements.map((e) => (e.id === selected.id ? { ...e, props: { ...e.props, [key]: value } } : e)),
   358→            }
   359→          : s,
   360→      ),
   361→    }))
   362→  }
   363→  function move(dir: -1 | 1) {
   364→    if (!selected || !screen) return
   365→    const idx = screen.elements.findIndex((e) => e.id === selected.id)
   366→    const to = idx + dir
   367→    if (to < 0 || to >= screen.elements.length) return
   368→    const els = [...screen.elements]
   369→    ;[els[idx], els[to]] = [els[to], els[idx]]
   370→    setDoc((d) => ({
   371→      ...d,
   372→      screens: d.screens.map((s) => (s.id === screen.id ? { ...s, elements: els } : s)),
   373→    }))
   374→  }
   375→  function duplicate() {
   376→    if (!selected || !screen) return
   377→    setDoc((d) => ({
   378→      ...d,
   379→      screens: d.screens.map((s) =>
   380→        s.id === screen.id ? { ...s, elements: [...s.elements, { ...selected, id: uid() }] } : s,
   381→      ),
   382→    }))
   383→  }
   384→  function remove() {
   385→    if (!selected || !screen) return
   386→    setDoc((d) => ({
   387→      ...d,
   388→      screens: d.screens.map((s) => (s.id === screen.id ? { ...s, elements: s.elements.filter((e) => e.id !== selected.id) } : s)),
   389→    }))
   390→    setSelected(null)
   391→  }
   392→
   393→  if (!selected) {
   394→    return (
   395→      <p className="rounded-xl border border-dashed p-4 text-center text-xs leading-relaxed text-muted-foreground">
   396→        Cliquez un élément de la maquette pour l’éditer, ou ajoutez-en un depuis le catalogue sous l’aperçu.
   397→      </p>
   398→    )
   399→  }
   400→
   401→  return (
   402→    <div className="space-y-3">
   403→      <div className="flex items-center justify-between">
   404→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   405→          {elLabel(selected.type)}
   406→        </p>
   407→        <button onClick={() => setSelected(null)} className="text-muted-foreground hover:text-foreground" aria-label="Désélectionner">
   408→          <X className="h-3.5 w-3.5" />
   409→        </button>
   410→      </div>
   411→      {PROPERTY_FIELDS[selected.type].map((f) =>
   412→        f.options ? (
   413→          <label key={f.key} className="flex items-center justify-between gap-2 text-xs">
   414→            <span className="shrink-0">{f.label}</span>
   415→            <select
   416→              value={selected.props[f.key] ?? f.options[0]}
   417→              onChange={(e) => patchProp(f.key, e.target.value)}
   418→              className="h-8 w-36 rounded-md border bg-background px-2 text-xs"
   419→              aria-label={f.label}
   420→            >
   421→              {f.options.map((o) => (
   422→                <option key={o} value={o}>{o}</option>
   423→              ))}
   424→            </select>
   425→          </label>
   426→        ) : f.multiline ? (
   427→          <label key={f.key} className="block space-y-1 text-xs">
   428→            <span>{f.label}</span>
   429→            <Textarea
   430→              value={selected.props[f.key] ?? ''}
   431→              onChange={(e) => patchProp(f.key, e.target.value)}
   432→              className="min-h-16 text-xs"
   433→              aria-label={f.label}
   434→            />
   435→          </label>
   436→        ) : (
   437→          <label key={f.key} className="block space-y-1 text-xs">
   438→            <span>{f.label}</span>
   439→            <Input
   440→              value={selected.props[f.key] ?? ''}
   441→              onChange={(e) => patchProp(f.key, e.target.value)}
   442→              className="h-8 text-xs"
   443→              aria-label={f.label}
   444→            />
   445→          </label>
   446→        ),
   447→      )}
   448→      {PROPERTY_FIELDS[selected.type].length === 0 && (
   449→        <p className="text-xs text-muted-foreground">Cet élément n’a pas de propriété.</p>
   450→      )}
   451→      <div className="flex flex-wrap gap-1.5 border-t pt-2.5">
   452→        <Button size="sm" variant="outline" className="h-7 gap-1 text-[11px]" onClick={() => move(-1)} aria-label="Monter l’élément">
   453→          <ArrowUp className="h-3 w-3" /> Monter
   454→        </Button>
   455→        <Button size="sm" variant="outline" className="h-7 gap-1 text-[11px]" onClick={() => move(1)} aria-label="Descendre l’élément">
   456→          <ArrowDown className="h-3 w-3" /> Descendre
   457→        </Button>
   458→        <Button size="sm" variant="outline" className="h-7 gap-1 text-[11px]" onClick={duplicate} aria-label="Dupliquer l’élément">
   459→          <Plus className="h-3 w-3" /> Dupliquer
   460→        </Button>
   461→        <Button size="sm" variant="outline" className="h-7 gap-1 text-[11px] text-destructive" onClick={remove} aria-label="Supprimer l’élément">
   462→          <Trash2 className="h-3 w-3" /> Supprimer
   463→        </Button>
   464→      </div>
   465→    </div>
   466→  )
   467→}

   469→/* ------------------------------------------------------------------ */
   470→/* Guide pas à pas (assistant à chacun des pas)                        */
   471→/* ------------------------------------------------------------------ */
   472→
   473→function nextStep(doc: DesignDoc): { title: string; hint: string } {
   474→  const avgEls = doc.screens.reduce((n, s) => n + s.elements.length, 0) / Math.max(1, doc.screens.length)
   475→  if (doc.screens.length < 2)
   476→    return { title: 'Structurez l’arborescence', hint: 'Ajoutez un second écran (ex. : Détail, Profil, Contact) pour dessiner la navigation.' }
   477→  if (avgEls < 4)
   478→    return { title: 'Enrichissez les écrans', hint: 'Chaque écran gagne à avoir 4 à 10 éléments : héro, texte, liste, carte, bouton d’action.' }
   479→  if (doc.palette.primary === DEFAULT_PALETTE.primary)
   480→    return { title: 'Personnalisez la palette', hint: 'Choisissez une couleur principale et un fond qui racontent votre marque (contraste AA).' }
   481→  return { title: 'Exportez le code', hint: 'Votre maquette est prête : onglet Code → choisissez HTML, React ou FlutterFlow.' }
   482→}
   483→
   484→function StepsGuide({ doc }: { doc: DesignDoc }) {
   485→  const checks = [
   486→    { label: 'Nom & identité définis', ok: doc.appName.trim().length > 2 && doc.appName !== 'Mon application' },
   487→    { label: 'Au moins 2 écrans', ok: doc.screens.length >= 2 },
   488→    { label: 'Écrans garnis (≥ 4 éléments en moyenne)', ok: doc.screens.reduce((n, s) => n + s.elements.length, 0) / Math.max(1, doc.screens.length) >= 4 },
   489→    { label: 'Palette personnalisée', ok: doc.palette.primary !== DEFAULT_PALETTE.primary },
   490→  ]
   491→  const step = nextStep(doc)
   492→  return (
   493→    <div className="space-y-2">
   494→      <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Progression</p>
   495→      {checks.map((c) => (

   533→  async function ask(q: string) {
   534→    const question = q.trim()

   536→    setBusy(true)
   537→    setInput('')
   538→    setLog((l) => [...l, { role: 'user', content: question }])

   541→        'Tu es l’assistant de conception WYSIWYG de Lumo. Voici le design actuel (JSON) :\n' +

   543→        '\n\nL’utilisateur demande : « ' + question + ' ».\n' +
   544→        'Modifie le design en conséquence et réponds en DEUX parties :\n' +
   545→        '1. Une ou deux phrases sur ce que tu as changé.\n' +
   546→        '2. Un UNIQUE bloc ```json avec le design COMPLET mis à jour — même schéma, types autorisés : navbar, hero, text, button, input, image, list, card, badge, divider, footer. Rien d’autre.'

   583→            <span className="mb-0.5 block font-semibold">{l.role === 'user' ? 'Vous' : 'Lumo · assistant design'}</span>
   584→            {l.content}
   585→          </p>
   586→        ))}
   587→        {busy && <p className="px-2 text-[11px] text-primary">L’assistant réfléchit…</p>}
   588→      </div>
   589→      <div className="flex flex-wrap gap-1">
   590→        {QUICK_PROMPTS.map((q) => (
   591→          <button
   592→            key={q}
   593→            onClick={() => void ask(q)}
   594→            disabled={busy}
   595→            className="rounded-full border bg-background px-2 py-0.5 text-[10px] text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground disabled:opacity-50"
   596→          >
   597→            {q}
   598→          </button>
   599→        ))}
   600→      </div>
   601→      <div className="flex items-end gap-1.5">
   602→        <Textarea
   603→          value={input}
   604→          onChange={(e) => setInput(e.target.value)}
   605→          onKeyDown={(e) => {
   606→            if (e.key === 'Enter' && !e.shiftKey) {
   607→              e.preventDefault()

   727→export function DesignStudioDialog({
   728→  artifact,
   729→  open,
   730→  onOpenChange,
   731→}: {
   732→  artifact: Artifact | null
   733→  open: boolean
   734→  onOpenChange: (v: boolean) => void
   735→}) {
   736→  const { toast } = useToast()
   737→  const sessions = useLumo((s) => s.sessions)
   738→  const updateMessage = useLumo((s) => s.updateMessage)
   739→  const [doc, setDocState] = useState<DesignDoc>(() => emptyDoc())
   740→  const [activeScreenId, setActiveScreenId] = useState(doc.screens[0]?.id ?? '')
   741→  const [selectedId, setSelectedId] = useState<string | null>(null)
   742→  const [viewport, setViewport] = useState<'desktop' | 'mobile'>('desktop')
   743→  const [loadedKey, setLoadedKey] = useState<string | null>(null)
   744→
   745→  /* Chargement d’un artefact — ajustement de état pendant le rendu (pattern React officiel) */
   746→  const loadKey = open && artifact ? artifact.id : null
   747→  if (loadKey && loadKey !== loadedKey) {
   748→    let next: DesignDoc | null = null
   749→    if (artifact?.text) {
   750→      try {
   751→        next = coerceDoc(JSON.parse(artifact.text))
   752→      } catch {
   753→        next = null
   754→      }
   755→    }
   756→    setDocState(next ?? emptyDoc())
   757→    setActiveScreenId(next?.screens[0]?.id ?? '')
   758→    setSelectedId(null)
   759→    setLoadedKey(loadKey)
   760→  }
   761→
   762→  const setDoc = (fn: (d: DesignDoc) => DesignDoc) => setDocState((d) => fn(d))
   763→  const activeScreen = doc.screens.find((s) => s.id === activeScreenId) ?? doc.screens[0]
   764→  const selected = activeScreen?.elements.find((e) => e.id === selectedId) ?? null
   765→
   766→  function addElement(type: ElType) {
   767→    if (!activeScreen) return
   768→    const el = makeEl(type)
   769→    setDoc((d) => ({
   770→      ...d,
   771→      screens: d.screens.map((s) => (s.id === activeScreen.id ? { ...s, elements: [...s.elements, el] } : s)),
   772→    }))
   773→    setSelectedId(el.id)
   774→  }
   775→
   776→  function renameScreen(name: string) {
   777→    if (!activeScreen) return
   778→    setDoc((d) => ({ ...d, screens: d.screens.map((s) => (s.id === activeScreen.id ? { ...s, name: name.slice(0, 40) } : s)) }))
   779→  }
   780→
   781→  function saveToArtifact() {
   782→    if (!artifact) return
   783→    for (const sess of sessions) {
   784→      const msg = sess.messages.find((m) => m.artifacts?.some((a) => a.id === artifact.id))
   785→      if (msg) {
   786→        updateMessage(sess.id, msg.id, {
   787→          artifacts: msg.artifacts?.map((a) =>
   788→            a.id === artifact.id
   789→              ? {
   790→                  ...a,
   791→                  text: JSON.stringify(doc),
   792→                  meta: { ...a.meta, appName: doc.appName, screens: doc.screens.length, elements: doc.screens.reduce((n, s) => n + s.elements.length, 0) },
   793→                }
   794→              : a,
   795→          ),
   796→        })
   797→        toast({ title: 'Maquette enregistrée', description: 'Les modifications sont conservées dans la session.' })
   798→        return
   799→      }
   800→    }
   801→    toast({ title: 'Enregistrement local', description: 'Maquette conservée jusqu’à la fermeture du studio.' })
   802→  }
   803→
   804→  const canvas = activeScreen && (
   805→    <div className="flex min-h-0 flex-1 flex-col">
   806→      <div className="flex flex-wrap items-center justify-between gap-2 px-1 pb-2">
   807→        <div className="flex min-w-0 items-center gap-1.5">
   808→          <Input
   809→            value={activeScreen.name}
   810→            onChange={(e) => renameScreen(e.target.value)}
   811→            className="h-8 w-40 text-xs font-semibold sm:w-52"
   812→            aria-label="Nom de l’écran"
   813→          />
   814→          <span className="text-[10px] text-muted-foreground">{activeScreen.elements.length} éléments</span>
   815→        </div>
   816→        <div className="flex rounded-lg border bg-background p-0.5" aria-label="Taille de la maquette">
   817→          <button onClick={() => setViewport('mobile')} aria-pressed={viewport === 'mobile'} aria-label="Maquette mobile 390 px" className={`rounded-md p-1.5 ${viewport === 'mobile' ? 'bg-primary text-primary-foreground' : 'text-muted-foreground'}`}>
   818→            <Smartphone className="h-3.5 w-3.5" />
   819→          </button>
   820→          <button onClick={() => setViewport('desktop')} aria-pressed={viewport === 'desktop'} aria-label="Maquette bureau" className={`rounded-md p-1.5 ${viewport === 'desktop' ? 'bg-primary text-primary-foreground' : 'text-muted-foreground'}`}>
   821→            <Monitor className="h-3.5 w-3.5" />
   822→          </button>
   823→        </div>
   824→      </div>
   825→      <div className="min-h-0 flex-1 overflow-y-auto rounded-2xl border bg-secondary/50 p-3 scrollbar-thin">
   826→        <div
   827→          className={`mx-auto flex min-h-full flex-col gap-3 rounded-3xl border p-3 shadow-sm transition-all ${
   828→            viewport === 'mobile' ? 'w-full max-w-[390px]' : 'w-full'

   830→          style={{ background: doc.palette.bg, color: doc.palette.text }}
   831→        >
   832→          {activeScreen.elements.length === 0 && (
   833→            <p className="py-10 text-center text-xs text-muted-foreground">Écran vide — ajoutez un élément ci-dessous.</p>
   834→          )}
   835→          {activeScreen.elements.map((el) => (
   836→            <ElView key={el.id} el={el} palette={doc.palette} selected={el.id === selectedId} onSelect={() => setSelectedId(el.id)} />
   837→          ))}
   838→        </div>
   839→      </div>
   840→      {/* Catalogue d’éléments */}
   841→      <div className="mt-2 flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin" aria-label="Ajouter un élément">

   849→            <Plus className="h-3 w-3" aria-hidden="true" />

   859→      <DialogContent className="flex h-[94dvh] w-[min(97vw,1400px)] max-w-none! flex-col gap-0 overflow-hidden p-0 sm:h-[92dvh]">
   860→        <DialogHeader className="border-b px-4 py-3 sm:px-5">
   861→          <DialogTitle className="flex items-center gap-2 font-display text-base">
   862→            <LayoutDashboard className="h-4.5 w-4.5 text-primary" aria-hidden="true" />
   863→            Studio visuel — {doc.appName}
   864→          </DialogTitle>
   865→          <DialogDescription>
   866→            WYSIWYG responsive · assistant à chaque étape · export HTML, React ou FlutterFlow
   867→          </DialogDescription>
   868→        </DialogHeader>
   869→
   870→        {/* Bureau : 3 colonnes */}
   871→        <div className="hidden min-h-0 flex-1 gap-0 lg:grid lg:grid-cols-[230px_minmax(0,1fr)_320px]">

   875→              <StepsGuide doc={doc} />

   877→          </aside>
   878→          <section className="flex min-h-0 flex-col p-3">{canvas}</section>
