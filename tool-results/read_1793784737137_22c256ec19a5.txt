   177→function uid(): string {
   178→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
   179→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`
   180→}

   182→function truncate(s: string, n: number): string {
   183→  const t = s.trim().replace(/\s+/g, ' ')
   184→  return t.length <= n ? t : t.slice(0, n - 1) + '…'
   185→}

   187→/** Bloc texte des pièces jointes, injecté dans le prompt du modèle */
   188→function attachmentBlock(m: Message, maxPerFile = 12000): string {
   189→  if (!m.attachments?.length) return ''
   190→  const parts = m.attachments.map((a) =>
   191→    a.text
   192→      ? `--- Fichier joint : ${a.name} ---\n${a.text.slice(0, maxPerFile)}`
   193→      : `[Pièce jointe : ${a.name} (${a.type || 'fichier'}, ${(a.size / 1024).toFixed(0)} Ko)]`,
   194→  )
   195→  return '\n\n[Fichiers joints par l’utilisateur]\n' + parts.join('\n\n')
   196→}
   197→
   198→function attachmentSummary(attachments?: Attachment[]): string {
   199→  if (!attachments?.length) return ''
   200→  const names = attachments.map((a) => a.name).join(', ')
   201→  const texts = attachments
   202→    .filter((a) => a.text)
   203→    .map((a) => `--- ${a.name} ---\n${a.text!.slice(0, 2000)}`)
   204→    .join('\n\n')
   205→  return `\n\n(En tenant compte des fichiers joints : ${names})${texts ? '\n\n' + texts : ''}`
   206→}

   208→export const useLumo = create<LumoState>()(

   210→    (set, get) => {
   211→      function updateSession(id: string, fn: (s: Session) => Session) {
   212→        set((st) => ({ sessions: st.sessions.map((s) => (s.id === id ? fn(s) : s)) }))
   213→      }
   214→      function updateMessage(sessionId: string, msgId: string, patch: Partial<Message>) {
   215→        updateSession(sessionId, (s) => ({
   216→          ...s,
   217→          updatedAt: Date.now(),
   218→          messages: s.messages.map((m) => (m.id === msgId ? { ...m, ...patch } : m)),
   219→        }))
   220→      }
   221→
   222→      function addMessage(sessionId: string, msg: Message) {
   223→        updateSession(sessionId, (s) => ({ ...s, messages: [...s.messages, msg] }))
   224→      }
   225→
   226→      function revealStream(sessionId: string, msgId: string, full: string): Promise<string> {
   227→        return new Promise((resolve) => {
   228→          let revealed = 0
   229→          const chunk = Math.max(3, Math.ceil(full.length / 160))
   230→          const timer = setInterval(() => {
   231→            revealed = Math.min(full.length, revealed + chunk)
   232→            updateMessage(sessionId, msgId, { content: full.slice(0, revealed) })
   233→            if (revealed >= full.length) {
   234→              clearInterval(timer)
   235→              streams.delete(msgId)
   236→              updateMessage(sessionId, msgId, { content: full, pending: false })
   237→              resolve(full)
   238→            }
   239→          }, 16)
   240→          streams.set(msgId, {
   241→            timer,
   242→            sessionId,
   243→            resolve: (content) => {
   244→              updateMessage(sessionId, msgId, { content, pending: false, stopped: true })
   245→              resolve(content)
   246→            },
   247→          })
   248→        })
   249→      }
   250→
   251→      function later(sessionId: string, ms: number, fn: () => void) {
   252→        const t = setTimeout(() => {
   253→          const set_ = sessionTimers.get(sessionId)
   254→          if (set_) set_.delete(t)
   255→          fn()
   256→        }, ms)
   257→        if (!sessionTimers.has(sessionId)) sessionTimers.set(sessionId, new Set())
   258→        sessionTimers.get(sessionId)!.add(t)
   259→      }

   263→      async function chatFlow(
   264→        sessionId: string,
   265→        history: Message[],
   266→        override?: { provider?: string; hfModel?: string },
   267→      ) {
   268→        const session = get().sessions.find((s) => s.id === sessionId)
   269→        if (!session) return
   270→        const providerLabel =
   271→          override?.provider === 'hf'
   272→            ? `hf:${override.hfModel}`
   273→            : (override?.provider ?? session.models.text)
   274→        const assistantId = uid()
   275→        addMessage(sessionId, {
   276→          id: assistantId,
   277→          role: 'assistant',
   278→          content: '',
   279→          createdAt: Date.now(),
   280→          pending: true,
   281→          flow: 'chat',
   282→          provider: providerLabel,
   283→        })
   284→        try {
   285→          const turns = history
   286→            .filter((m) => !m.pending && !m.error && m.content.trim())
   287→            .slice(-16)
   288→            .map((m) => ({ role: m.role, content: m.content + attachmentBlock(m) }) as { role: 'user' | 'assistant'; content: string })
   289→          let system = LUMO_SYSTEM
   290→          if (session.mode === 'project') system = PROJECT_SYSTEM
   291→          if (session.mode === 'book' && session.brief) {
   292→            const bibleArtifact = session.messages
   293→              .flatMap((m) => m.artifacts ?? [])
   294→              .find((a) => a.kind === 'bible')
   295→            system = bookSystem(
   296→              bibleArtifact?.text ??
   297→                'Brief : ' + JSON.stringify(session.brief, null, 2),
   298→            )
   299→          }
   300→          const useOverride: ChatOverride | undefined =
   301→            override?.provider && override.provider !== 'lumo-core'
   302→              ? { provider: override.provider as ChatOverride['provider'], hfModel: override.hfModel }
   303→              : undefined
   304→          const full = await apiChat(turns, system, useOverride)
   305→          const stillPending = get().sessions
   306→            .find((s) => s.id === sessionId)
   307→            ?.messages.find((m) => m.id === assistantId)?.pending
   308→          if (stillPending) await revealStream(sessionId, assistantId, full)
   309→          else updateMessage(sessionId, assistantId, { content: full, pending: false })
   310→        } catch (e) {
   311→          const err = e as Error & { fallback?: string[] }
   312→          updateMessage(sessionId, assistantId, {
   313→            pending: false,
   314→            error: err.message,
   315→            content: '',
   316→            fallback: err.fallback ?? ['mistral', 'agnes', 'hf'],
   317→          })
   318→        }
   319→      }

   321→      async function imageFlow(sessionId: string, prompt: string, opts: SendOptions) {
   322→        const session = get().sessions.find((s) => s.id === sessionId)
   323→        if (!session) return
   324→        const count = Math.max(1, Math.min(4, opts.variations ?? 1))
   325→        const size = opts.imageSize ?? '1024x1024'
   326→        const provider = session.models.image
   327→        const fullPrompt = prompt + attachmentSummary(opts.attachments)
   328→        const assistantId = uid()
   329→        addMessage(sessionId, {
   330→          id: assistantId,
   331→          role: 'assistant',
   332→          content: '',
   333→          createdAt: Date.now(),
   334→          pending: true,
   335→          flow: 'image',
   336→          provider,
   337→        })
   338→        const results = await Promise.allSettled(
   339→          Array.from({ length: count }, (_, i) =>
   340→            i === 0
   341→              ? apiImage(fullPrompt, size)
   342→              : new Promise<{ base64: string; bytes: number }>((resolve, reject) =>
   343→                  setTimeout(() => apiImage(fullPrompt, size).then(resolve, reject), i * 3000),
   344→                ),
   345→          ),
   346→        )
   347→        const artifacts: Artifact[] = []
   348→        let ok = 0
   349→        for (let i = 0; i < results.length; i++) {
   350→          const r = results[i]
   351→          if (r.status === 'fulfilled') {
   352→            ok++
   353→            const a: Artifact = {
   354→              id: uid(),
   355→              kind: 'image',
   356→              name: `image_${String(artifacts.length + 1).padStart(3, '0')}.png`,
   357→              mimeType: 'image/png',
   358→              createdAt: Date.now(),
   359→              provider,
   360→              prompt: fullPrompt,
   361→              data: r.value.base64,
   362→              bytes: r.value.bytes,
   363→              meta: { size },
   364→            }
   365→            void putArtifactData(a.id, a.data)
   366→            artifacts.push(a)
   367→          }
   368→        }
   369→        const failed = count - ok
   370→        const content =
   371→          ok === 0
   372→            ? `La génération d’image a échoué pour « ${truncate(prompt, 60)} ». Réessayez dans un instant.`
   373→            : `Voici ${ok > 1 ? `les ${ok} variations` : 'votre visuel'} pour « ${truncate(prompt, 60)} ».${
   374→                failed ? ` (${failed} tentative(s) échouée(s))` : ''
   375→              }`
   376→        updateMessage(sessionId, assistantId, {
   377→          content,
   378→          pending: false,
   379→          artifacts,
   380→          error: ok === 0 ? 'Génération impossible' : undefined,
   381→        })
   382→      }

   606→          const err = e as Error & { fallback?: string[] }
   607→          updateMessage(sessionId, assistantId, {
   608→            pending: false,
   609→            error: `Génération du code impossible : ${err.message}`,

   611→            fallback: err.fallback ?? ['mistral', 'agnes', 'hf'],
   612→          })
   613→        }
