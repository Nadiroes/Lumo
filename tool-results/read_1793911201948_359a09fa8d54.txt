   133→/** Nettoie un lrc parsé : « \\n » littéraux → retours ligne, backslashes parasites, espaces. */
   134→function sanitizeLrc(raw: string): string {
   135→  return raw
   136→    .replace(/\\r/g, '')
   137→    .replace(/\\n/g, '\n') // « \n » écrit littéralement par le modèle
   138→    .replace(/\\(?!["\\/bfnrtu])/g, '') // backslash isolé parasite
   139→    .split('\n')
   140→    .map((l) => l.replace(/\s+$/, ''))
   141→    .join('\n')
   142→    .replace(/\n{3,}/g, '\n\n')
   143→    .trim()
   144→}

   146→/**
   147→ * Parse la réponse du parolier avec tolérance :
   148→ * 1. JSON strict extrait du texte ; 2. réparation des échappements invalides
   149→ * et caractères de contrôle ; 3. extraction par regex du champ lrc en dernier recours.
   150→ */
   151→function tolerantParseLyrics(raw: string): { title: string; lrc: string } | undefined {
   152→  const start = raw.indexOf('{')
   153→  const end = raw.lastIndexOf('}')
   154→  if (start < 0 || end <= start) return undefined
   155→  const chunk = raw.slice(start, end + 1)
   156→
   157→  try {
   158→    const parsed = JSON.parse(chunk) as { title?: string; lrc?: string }
   159→    if (parsed?.lrc) return { title: String(parsed.title ?? ''), lrc: String(parsed.lrc) }
   160→  } catch {
   161→    /* réparation ci-dessous */
   162→  }
   163→
   164→  try {
   165→    const repaired = chunk
   166→      .replace(/\\(?!["\\/bfnrtu])/g, '\\\\') // \x invalide → \\x
   167→      .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, ' ') // contrôles bruts
   168→    const parsed = JSON.parse(repaired) as { title?: string; lrc?: string }
   169→    if (parsed?.lrc) return { title: String(parsed.title ?? ''), lrc: String(parsed.lrc) }
   170→  } catch {
   171→    /* extraction regex ci-dessous */
   172→  }
   173→
   174→  const lrcMatch = chunk.match(/"lrc"\s*:\s*"([\s\S]*)/) // jusqu'à la fin, on déséchappe à la main
   175→  if (lrcMatch) {
   176→    let body = lrcMatch[1]
   177→    const lastQuote = body.lastIndexOf('"')
   178→    if (lastQuote > 0) body = body.slice(0, lastQuote)
   179→    const unescaped = body
   180→      .replace(/\\"/g, '"')
   181→      .replace(/\\\\/g, '\\')
   182→      .replace(/\\n/g, '\n')
   183→    const titleMatch = chunk.match(/"title"\s*:\s*"([^"]*)"/)
   184→    return { title: titleMatch?.[1] ?? '', lrc: unescaped }
   185→  }
   186→  return undefined
   187→}

   189→function parseLyricsJson(raw: string, what: string): LyricsResult {
   190→  const parsed = tolerantParseLyrics(raw)
   191→  if (!parsed) throw new Error(`${what} : réponse illisible`)
   192→  const lrc = sanitizeLrc(parsed.lrc)
   193→  if (!lrc || !hasV2Sections(lrc)) throw new Error(`${what} : sections absentes`)
   194→  return { title: parsed.title.trim() || 'Chanson Lumo', lrc }
   195→}

   311→/** Durée réelle du fichier via ffprobe (l'estimation par débit binaire est fausse en VBR). */
   312→async function probeDuration(file: string): Promise<number> {
   313→  const { stdout } = await runTool(
   314→    'ffprobe',
   315→    ['-v', 'error', '-show_entries', 'format=duration', '-of', 'json', file],
   316→    { timeout: 15_000 },
   317→  )
   318→  const d = Number(JSON.parse(stdout)?.format?.duration)
   319→  return Number.isFinite(d) && d > 0 ? d : 0
   320→}
   321→
   322→/**
   323→ * « Écoute honnête » :
   324→ * 1. mesure la durée réelle (ffprobe) ;
   325→ * 2. si le morceau dépasse nettement la durée demandée — le Space v2 rend toujours ~240 s,
   326→ *    quitte à enchainer 2 minutes de jam instrumental après la fin du chant — on coupe à la
   327→ *    durée cible avec un fondu de sortie de 3 s (pas de coupure brutale au milieu d'un temps fort) ;
   328→ * 3. jamais bloquant : ffmpeg absent ou en échec → fichier original conservé tel quel.
   329→ */
   330→async function refineAudio(
   331→  base64: string,
   332→  mime: string,
   333→  targetSec: number,
   334→): Promise<{ data: Buffer; durationSec: number; trimmed: boolean }> {
   335→  const ext = mime === 'audio/ogg' ? 'ogg' : mime === 'audio/wav' ? 'wav' : 'mp3'
   336→  const input = Buffer.from(base64, 'base64')
   337→  let real = 0
   338→  const dir = await mkdtemp(join(tmpdir(), 'lumo-audio-'))
   339→  const src = join(dir, `in.${ext}`)
   340→  const out = join(dir, `out.${ext}`)
   341→  try {
   342→    await writeFile(src, input)
   343→    try {
   344→      real = await probeDuration(src)
   345→    } catch {
   346→      return { data: input, durationSec: 0, trimmed: false }
   347→    }
   348→    const durationSec = Math.round(real)
   349→    const target = Math.max(30, Math.min(285, Math.round(targetSec)))
   350→    if (real <= target + 4) return { data: input, durationSec, trimmed: false }
   351→
   352→    const fadeStart = Math.max(0, target - 3)
   353→    const fadeDur = Math.min(3, target - fadeStart)
   354→    const args = [
   355→      '-v', 'error', '-y', '-i', src,
   356→      '-t', String(target),
   357→      '-af', `afade=t=in:st=0:d=0.05,afade=t=out:st=${fadeStart}:d=${fadeDur}`,
   358→    ]
   359→    if (ext === 'mp3') args.push('-c:a', 'libmp3lame', '-b:a', '192k')
   360→    else if (ext === 'ogg') args.push('-c:a', 'libvorbis', '-q:a', '6')
   361→    args.push(out)
   362→    await runTool('ffmpeg', args, { timeout: 60_000 })
   363→    const data = await readFile(out)
   364→    if (data.byteLength < 1000) throw new Error('découpe vide')
   365→    return { data, durationSec: target, trimmed: true }
   366→  } catch {
   367→    return { data: input, durationSec: Math.round(real) || 0, trimmed: false }
   368→  } finally {
   369→    void rm(dir, { recursive: true, force: true }).catch(() => {})
   370→  }
   371→}

   373→/**
   374→ * Appel Space avec reconnexion automatique (nouvelle session Gradio à chaque essai).
   375→ * « 404: Session not found » = session périmée : le Space ZeroGPU a redémarré entre
   376→ * la connexion et l'appel — une reconnexion fraîche suffit presque toujours.
   377→ */
   378→async function predictWithRetry(
   379→  connect: () => Promise<GradioClient>,
   380→  endpoint: string,
   381→  params: Record<string, unknown>,
   382→  timeoutMs = 240_000,
   383→  tries = 3,
   384→): Promise<{ data: unknown }> {
   385→  let lastErr: unknown
   386→  for (let i = 0; i < tries; i++) {
   387→    try {
   388→      const client = await connect()
   389→      return (await Promise.race([
   390→        client.predict(endpoint, params),
   391→        new Promise<never>((_, rej) =>
   392→          setTimeout(() => rej(new Error(`Délai du studio dépassé (${Math.round(timeoutMs / 1000)} s)`)), timeoutMs),
   393→        ),
   394→      ])) as { data: unknown }
   395→    } catch (e) {
   396→      lastErr = e
   397→      const msg = e instanceof Error ? e.message : String(e)
   398→      const transient =
   399→        /session not found|connection closed|fetch failed|network|econnreset|socket|502|503|overloaded/i.test(msg)
   400→      if (!transient || i === tries - 1) throw e
   401→      await new Promise((r) => setTimeout(r, 1500 * (i + 1)))
   402→    }
   403→  }
   404→  throw lastErr
   405→}

   430→/**
   431→ * Paramètres du Space DiffRhythm 1 : convertit les paroles à sections en LRC
   432→ * horodaté (~4,2 s par ligne, chant régulier), durée plafonnée à 95 s.
   433→ */
   434→function buildV1Params(lrc: string, body: MusicBody): Record<string, unknown> {
   435→  const lines = lrc
   436→    .split('\n')
   437→    .map((l) => l.trim())
   438→    .filter((l) => l && !l.startsWith('['))
   439→  const START = 12
   440→  const PER_LINE = 4.2
   441→  let t = START
   442→  const stamped = lines.map((line) => {
   443→    const mm = String(Math.floor(t / 60)).padStart(2, '0')
   444→    const ss = (t % 60).toFixed(3).padStart(6, '0')
   445→    const out = `[${mm}:${ss}] ${line}`
   446→    t += PER_LINE
   447→    return out
   448→  })
   449→  const duration = Math.max(30, Math.min(95, Math.round(START + lines.length * PER_LINE + 8)))
   450→  return {
   451→    /* v1 chante chaque ligne : en instrumental on laisse une seule ligne de vocalise minimale */
   452→    lrc: body.instrumental && stamped.length === 0 ? '[00:12.000] ♪' : stamped.join('\n'),
   453→    ref_audio_path: null,
   454→    text_prompt: buildTextPrompt(body) || (body.instrumental ? 'instrumental' : 'pop, piano, warm vocals'),
   455→    seed: 0,
   456→    randomize_seed: true,
   457→    steps: 32,
   458→    cfg_strength: 4.0,
   459→    file_type: body.format ?? 'mp3',
   460→    odeint_method: 'euler',
   461→    preference_infer: 'quality first',
   462→    Music_Duration: duration,
   463→  }
   464→}

   466→/** Appel du Space DiffRhythm 2 (avec ou sans token HF) et téléchargement du morceau. */
   467→async function callSpaceV2(
   468→  lrc: string,
   469→  body: MusicBody,
   470→  token?: `hf_${string}`,
   471→): Promise<AudioResult> {
   472→  const { Client } = await import('@gradio/client')
   473→  const format = body.format ?? 'mp3'
   474→
   475→  const result = await predictWithRetry(
   476→    () => Client.connect(SPACE_V2, token ? { token } : undefined) as Promise<GradioClient>,
   477→    '/infer_music',
   478→    buildV2Params(lrc, body),
   479→  )
   480→
   481→  const first = Array.isArray(result.data) ? result.data[0] : result.data
   482→  const url = (first as { url?: string })?.url
   483→  if (!url) throw new Error('Le studio n’a pas renvoyé de fichier audio')
   484→
   485→  const dl = await fetch(url, { signal: AbortSignal.timeout(90_000) })
   486→  if (!dl.ok) throw new Error(`Téléchargement du morceau impossible (${dl.status})`)
   487→  const buf = await dl.arrayBuffer()
   488→  return {
   489→    base64: Buffer.from(buf).toString('base64'),
   490→    bytes: buf.byteLength,
   491→    mime: format === 'mp3' ? 'audio/mpeg' : format === 'ogg' ? 'audio/ogg' : 'audio/wav',
   492→    engine: token ? 'DiffRhythm 2 · compte HF' : 'DiffRhythm 2 · accès public',
   493→  }
   494→}

   525→export async function POST(req: NextRequest) {
   526→  const enc = new TextEncoder()
   527→  let body: MusicBody
   528→  try {
   529→    body = (await req.json()) as MusicBody
   530→  } catch {
   531→    return new Response(JSON.stringify({ ok: false, error: 'Corps de requête invalide' }), { status: 400 })
   532→  }
   533→  const prompt = (body.prompt ?? '').trim()
   534→  const userLyrics = (body.lyrics ?? '').trim()
   535→  if (!prompt && !userLyrics) {
   536→    return new Response(
   537→      JSON.stringify({ ok: false, error: 'Décrivez au moins le thème de la chanson' }),
   538→      { status: 400 },
   539→    )
   540→  }
   541→
   542→  const stream = new ReadableStream<Uint8Array>({
   543→    async start(controller) {
   544→      let closed = false
   545→      const safeEmit = (obj: unknown) => {
   546→        if (!closed) controller.enqueue(enc.encode(JSON.stringify(obj) + '\n'))
   547→      }
   548→      try {
   549→        /* ---- 1. Paroles ---- */
   550→        let title: string
   551→        let lrc: string
   552→        if (body.instrumental) {
   553→          /* Instrumental : pas de parolier — LRC minimal, le prompt « no vocals » dirige le style.
   554→           * DiffRhythm 2 génère alors une pièce musicale sans voix. */
   555→          title = fallbackTitle(body, '')
   556→          lrc = '[start]'
   557→          safeEmit({ stage: 'lyrics', source: 'instrumental', title, lrc })
   558→        } else if (userLyrics && hasV2Sections(userLyrics)) {
   559→          title = fallbackTitle(body, userLyrics)
   560→          lrc = userLyrics.startsWith('[start]') ? userLyrics : `[start]\n${userLyrics}`
   561→          safeEmit({ stage: 'lyrics', source: 'user', title, lrc })
   562→        } else {
   563→          try {
   564→            const res = userLyrics
   565→              ? await structureUserLyrics(body, cleanUserLyrics(userLyrics))
   566→              : await generateLyrics(body)
   567→            title = res.title
   568→            lrc = res.lrc
   569→            safeEmit({ stage: 'lyrics', source: userLyrics ? 'user-structured' : 'glm', title, lrc })
   570→          } catch (e) {
   571→            if (userLyrics) {
   572→              title = fallbackTitle(body, userLyrics)
   573→              lrc = `[start]\n[verse]\n${cleanUserLyrics(userLyrics)}`
   574→              safeEmit({ stage: 'lyrics', source: 'user-raw', title, lrc })
   575→            } else {
   576→              throw e
   577→            }
   578→          }
   579→        }
   580→
   581→        /* ---- 2. Studio DiffRhythm — chaîne de repli (quota ZeroGPU ≠ par compte) ----
   582→         *  a) DiffRhythm 2 avec le token HF de l'utilisateur (quota personnel)
   583→         *  b) DiffRhythm 2 en accès public (seau de quota anonyme, distinct)
   584→         *  c) DiffRhythm 1 en accès public (95 s max, LRC horodaté)
   585→         * On ne bascule vers la synthèse locale qu'en dernier ressort. */
   586→        safeEmit({ stage: 'studio', engine: ENGINE })
   587→        const token = await getUserHfToken()
   588→        const attempts: { label: string; run: () => Promise<AudioResult> }[] = [
   589→          ...(token ? [{ label: 'votre compte Hugging Face', run: () => callSpaceV2(lrc, body, token) }] : []),
   590→          { label: 'l’accès public DiffRhythm 2', run: () => callSpaceV2(lrc, body) },
   591→          { label: 'DiffRhythm 1 (extraits ≤ 95 s)', run: () => callSpaceV1(lrc, body) },
   592→        ]
   593→
   594→        /* On commence directement par le chemin qui a fonctionné récemment
   595→         * (évite de refaire échouer les tentatives en quota mort à chaque génération). */
   596→        if (lastGoodAttempt > 0 && lastGoodAttempt < attempts.length && Date.now() - lastGoodAt < 10 * 60_000) {
   597→          const [good] = attempts.splice(lastGoodAttempt, 1)
   598→          attempts.unshift(good)
   599→        }
   600→
   601→        let audio: AudioResult | undefined
   602→        let lastReason = 'studio indisponible'
   603→        let refined: { durationSec: number; trimmed: boolean } | undefined
   604→        for (let i = 0; i < attempts.length; i++) {
   605→          const attempt = attempts[i]
   606→          try {
   607→            safeEmit({ stage: 'studio-attempt', label: attempt.label })
   608→            audio = await attempt.run()
   609→            lastGoodAttempt = i
   610→            lastGoodAt = Date.now()
   611→            /* Écoute honnête : durée réelle mesurée + découpe propre à la durée demandée
   612→             * (le Space v2 rend ~240 s même pour une chanson demandée à 1 ou 2 min). */
   613→            const targetSec = Math.min(Math.max(body.duration ?? 120, 30), 285)
   614→            const ref = await refineAudio(audio.base64, audio.mime, targetSec)
   615→            audio = {
   616→              ...audio,
   617→              base64: ref.data.toString('base64'),
   618→              bytes: ref.data.byteLength,
   619→            }
   620→            refined = { durationSec: ref.durationSec, trimmed: ref.trimmed }
   621→            break
   622→          } catch (e) {
   623→            const raw = e instanceof Error ? e.message : String(e)
   624→            lastReason = isQuotaError(raw)
   625→              ? `Quota GPU Hugging Face épuisé pour ${attempt.label}` + (quotaResetIn(raw) ? ` (réinitialisation dans ${quotaResetIn(raw)})` : '')
   626→              : /session not found/i.test(raw)
   627→                ? `session du studio expirée (${attempt.label}) — le Space a redémarré, nouvelles tentatives effectuées`
   628→                : raw.slice(0, 180)
   629→            safeEmit({ stage: 'studio-retry', label: attempt.label, reason: lastReason })
   630→          }
   631→        }
   632→
   633→        if (audio) {
   634→          const durationSec =
   635→            refined?.durationSec && refined.durationSec > 0
   636→              ? refined.durationSec
   637→              : audio.mime === 'audio/mpeg'
   638→                ? Math.round((audio.bytes * 8) / 128_000)
   639→                : 0
   640→          safeEmit({
   641→            stage: 'done',
   642→            engine: audio.engine,
   643→            title,
   644→            lrc,
   645→            base64: audio.base64,
   646→            bytes: audio.bytes,
   647→            mime: audio.mime,
   648→            durationSec,
   649→            quality: qualityPreset(body).label,
   650→            steps: qualityPreset(body).steps,
   651→            solver: qualityPreset(body).solver,
   652→            trimmed: refined?.trimmed === true,
   653→          })
   654→        } else {
   655→          /* Tous les chemins serveur sont morts (quotas IP serveur, Space en redémarrage).
   656→           * Dernière carte : le navigateur de l'utilisateur appelle lui-même le Space —
   657→           * sa connexion possède son propre seau de quota ZeroGPU, indépendant du serveur. */
   658→          safeEmit({
   659→            stage: 'client-studio',
   660→            reason: lastReason,
   661→            title,
   662→            lrc,
   663→            attempts: [
   664→              { space: SPACE_V2, label: 'DiffRhythm 2', params: buildV2Params(lrc, body) },
   665→              { space: SPACE_V1, label: 'DiffRhythm 1 (≤ 95 s)', params: buildV1Params(lrc, body) },
   666→            ],
   667→          })
   668→        }
   669→      } catch (e) {
   670→        safeEmit({
   671→          stage: 'fallback',
   672→          reason: e instanceof Error ? e.message.slice(0, 180) : 'erreur inattendue',
   673→        })
   674→      } finally {
   675→        closed = true
   676→        controller.close()
   677→      }
   678→    },
   679→  })
   680→
   681→  return new Response(stream, {
   682→    headers: {
   683→      'Content-Type': 'application/x-ndjson; charset=utf-8',
   684→      'Cache-Control': 'no-cache, no-transform',
   685→      'X-Accel-Buffering': 'no',
   686→    },
   687→  })
   688→}
