   107→function uid(): string {
   108→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
   109→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`
   110→}

   112→function truncate(s: string, n: number): string {
   113→  const t = s.trim().replace(/\s+/g, ' ')
   114→  return t.length <= n ? t : t.slice(0, n - 1) + '…'
   115→}

   117→/** Bloc texte des pièces jointes, injecté dans le prompt du modèle */
   118→function attachmentBlock(m: Message, maxPerFile = 12000): string {
   119→  if (!m.attachments?.length) return ''
   120→  const parts = m.attachments.map((a) =>
   121→    a.text
   122→      ? `--- Fichier joint : ${a.name} ---\n${a.text.slice(0, maxPerFile)}`
   123→      : `[Pièce jointe : ${a.name} (${a.type || 'fichier'}, ${(a.size / 1024).toFixed(0)} Ko)]`,
   124→  )
   125→  return '\n\n[Fichiers joints par l’utilisateur]\n' + parts.join('\n\n')
   126→}
   127→
   128→function attachmentSummary(attachments?: Attachment[]): string {
   129→  if (!attachments?.length) return ''
   130→  const names = attachments.map((a) => a.name).join(', ')
   131→  const texts = attachments
   132→    .filter((a) => a.text)
   133→    .map((a) => `--- ${a.name} ---\n${a.text!.slice(0, 2000)}`)
   134→    .join('\n\n')
   135→  return `\n\n(En tenant compte des fichiers joints : ${names})${texts ? '\n\n' + texts : ''}`
   136→}

   138→export const useLumo = create<LumoState>()(
   139→  persist(
   140→    (set, get) => {
   141→      function updateSession(id: string, fn: (s: Session) => Session) {
   142→        set((st) => ({ sessions: st.sessions.map((s) => (s.id === id ? fn(s) : s)) }))
   143→      }
   144→      function updateMessage(sessionId: string, msgId: string, patch: Partial<Message>) {
   145→        updateSession(sessionId, (s) => ({
   146→          ...s,
   147→          updatedAt: Date.now(),
   148→          messages: s.messages.map((m) => (m.id === msgId ? { ...m, ...patch } : m)),
   149→        }))
   150→      }
   151→
   152→      function addMessage(sessionId: string, msg: Message) {
   153→        updateSession(sessionId, (s) => ({ ...s, messages: [...s.messages, msg] }))
   154→      }
   155→
   156→      function revealStream(sessionId: string, msgId: string, full: string): Promise<string> {
   157→        return new Promise((resolve) => {
   158→          let revealed = 0
   159→          const chunk = Math.max(3, Math.ceil(full.length / 160))
   160→          const timer = setInterval(() => {
   161→            revealed = Math.min(full.length, revealed + chunk)
   162→            updateMessage(sessionId, msgId, { content: full.slice(0, revealed) })
   163→            if (revealed >= full.length) {
   164→              clearInterval(timer)
   165→              streams.delete(msgId)
   166→              updateMessage(sessionId, msgId, { content: full, pending: false })
   167→              resolve(full)
   168→            }
   169→          }, 16)
   170→          streams.set(msgId, {
   171→            timer,
   172→            sessionId,
   173→            resolve: (content) => {
   174→              updateMessage(sessionId, msgId, { content, pending: false, stopped: true })
   175→              resolve(content)
   176→            },
   177→          })
   178→        })
   179→      }
   180→
   181→      function later(sessionId: string, ms: number, fn: () => void) {
   182→        const t = setTimeout(() => {
   183→          const set_ = sessionTimers.get(sessionId)
   184→          if (set_) set_.delete(t)
   185→          fn()
   186→        }, ms)
   187→        if (!sessionTimers.has(sessionId)) sessionTimers.set(sessionId, new Set())
   188→        sessionTimers.get(sessionId)!.add(t)
   189→      }

   193→      async function chatFlow(
   194→        sessionId: string,
   195→        history: Message[],
   196→        override?: { provider?: string; hfModel?: string },
   197→      ) {
   198→        const session = get().sessions.find((s) => s.id === sessionId)
   199→        if (!session) return
   200→        const providerLabel =
   201→          override?.provider === 'hf'
   202→            ? `hf:${override.hfModel}`
   203→            : (override?.provider ?? session.models.text)
   204→        const assistantId = uid()
   205→        addMessage(sessionId, {
   206→          id: assistantId,
   207→          role: 'assistant',
   208→          content: '',
   209→          createdAt: Date.now(),
   210→          pending: true,
   211→          flow: 'chat',
   212→          provider: providerLabel,
   213→        })
   214→        try {
   215→          const turns = history
   216→            .filter((m) => !m.pending && !m.error && m.content.trim())
   217→            .slice(-16)
   218→            .map((m) => ({ role: m.role, content: m.content + attachmentBlock(m) }) as { role: 'user' | 'assistant'; content: string })
   219→          let system = LUMO_SYSTEM
   220→          if (session.mode === 'project') system = PROJECT_SYSTEM
   221→          if (session.mode === 'book' && session.brief) {
   222→            const bibleArtifact = session.messages
   223→              .flatMap((m) => m.artifacts ?? [])
   224→              .find((a) => a.kind === 'bible')
   225→            system = bookSystem(
   226→              bibleArtifact?.text ??
   227→                'Brief : ' + JSON.stringify(session.brief, null, 2),
   228→            )
   229→          }
   230→          const useOverride: ChatOverride | undefined =
   231→            override?.provider && override.provider !== 'lumo-core'
   232→              ? { provider: override.provider as ChatOverride['provider'], hfModel: override.hfModel }
   233→              : undefined
   234→          const full = await apiChat(turns, system, useOverride)
   235→          const stillPending = get().sessions
   236→            .find((s) => s.id === sessionId)
   237→            ?.messages.find((m) => m.id === assistantId)?.pending
   238→          if (stillPending) await revealStream(sessionId, assistantId, full)
   239→          else updateMessage(sessionId, assistantId, { content: full, pending: false })
   240→        } catch (e) {
   241→          const err = e as Error & { fallback?: string[] }
   242→          updateMessage(sessionId, assistantId, {
   243→            pending: false,
   244→            error: err.message,
   245→            content: '',
   246→            fallback: err.fallback ?? ['mistral', 'agnes', 'hf'],
   247→          })
   248→        }
   249→      }

   251→      async function imageFlow(sessionId: string, prompt: string, opts: SendOptions) {
   252→        const session = get().sessions.find((s) => s.id === sessionId)
   253→        if (!session) return
   254→        const count = Math.max(1, Math.min(4, opts.variations ?? 1))
   255→        const size = opts.imageSize ?? '1024x1024'
   256→        const provider = session.models.image
   257→        const fullPrompt = prompt + attachmentSummary(opts.attachments)
   258→        const assistantId = uid()
   259→        addMessage(sessionId, {
   260→          id: assistantId,
   261→          role: 'assistant',
   262→          content: '',
   263→          createdAt: Date.now(),
   264→          pending: true,
   265→          flow: 'image',
   266→          provider,
   267→        })
   268→        const results = await Promise.allSettled(
   269→          Array.from({ length: count }, (_, i) =>
   270→            i === 0
   271→              ? apiImage(fullPrompt, size)
   272→              : new Promise<{ base64: string; bytes: number }>((resolve, reject) =>
   273→                  setTimeout(() => apiImage(fullPrompt, size).then(resolve, reject), i * 3000),
   274→                ),
   275→          ),
   276→        )
   277→        const artifacts: Artifact[] = []
   278→        let ok = 0
   279→        for (let i = 0; i < results.length; i++) {
   280→          const r = results[i]
   281→          if (r.status === 'fulfilled') {
   282→            ok++
   283→            const a: Artifact = {
   284→              id: uid(),
   285→              kind: 'image',
   286→              name: `image_${String(artifacts.length + 1).padStart(3, '0')}.png`,
   287→              mimeType: 'image/png',
   288→              createdAt: Date.now(),
   289→              provider,
   290→              prompt: fullPrompt,
   291→              data: r.value.base64,
   292→              bytes: r.value.bytes,
   293→              meta: { size },
   294→            }
   295→            void putArtifactData(a.id, a.data)
   296→            artifacts.push(a)
   297→          }
   298→        }
   299→        const failed = count - ok
   300→        const content =
   301→          ok === 0
   302→            ? `La génération d’image a échoué pour « ${truncate(prompt, 60)} ». Réessayez dans un instant.`
   303→            : `Voici ${ok > 1 ? `les ${ok} variations` : 'votre visuel'} pour « ${truncate(prompt, 60)} ».${
   304→                failed ? ` (${failed} tentative(s) échouée(s))` : ''
   305→              }`
   306→        updateMessage(sessionId, assistantId, {
   307→          content,
   308→          pending: false,
   309→          artifacts,
   310→          error: ok === 0 ? 'Génération impossible' : undefined,
   311→        })
   312→      }

   422→      /** Génération musicale locale (façon Suno) — rendu WAV réel, lisible et téléchargeable */
   423→      async function musicFlow(sessionId: string, prompt: string, opts: SendOptions) {
   424→        const session = get().sessions.find((s) => s.id === sessionId)
   425→        if (!session) return
   426→        const provider = session.models.audio
   427→        const assistantId = uid()
   428→        addMessage(sessionId, {
   429→          id: assistantId,
   430→          role: 'assistant',
   431→          content: '🎼 Composition en cours… analyse du prompt, choix du style…',
   432→          createdAt: Date.now(),
   433→          pending: true,
   434→          flow: 'audio-music',
   435→          provider,
   436→        })
   437→        try {
   438→          later(sessionId, 500, () =>
   439→            updateMessage(sessionId, assistantId, {
   440→              content: '🎼 Style, tempo et tonalité choisis — enregistrement des pistes (nappes, basse, mélodie, batterie)…',
   441→            }),
   442→          )
   443→          const res = await generateMusic({
   444→            prompt,
   445→            style: opts.musicStyle ?? 'auto',
   446→            durationSec: opts.musicDuration ?? 24,
   447→            instrumental: opts.instrumental ?? false,
   448→          })
   449→          const artifact: Artifact = {
   450→            id: uid(),
   451→            kind: 'audio',
   452→            name: `${res.title.replace(/[^\wÀ-ÿ]+/g, '_')}.wav`,
   453→            mimeType: 'audio/wav',
   454→            createdAt: Date.now(),
   455→            provider,
   456→            prompt,
   457→            data: res.wavBase64,
   458→            bytes: res.bytes,
   459→            meta: {
   460→              music: true,
   461→              style: res.meta.style,
   462→              bpm: res.meta.bpm,
   463→              key: res.meta.key,
   464→              durationSec: res.meta.durationSec,
   465→              instrumental: opts.instrumental ?? false,
   466→            },
   467→          }
   468→          void putArtifactData(artifact.id, artifact.data!)
   469→          updateMessage(sessionId, assistantId, {
   470→            content:
   471→              `Voici **« ${res.title} »** — ${res.meta.style}, ${res.meta.bpm} BPM, ${res.meta.key} (${res.meta.durationSec} s).\n\n` +
   472→              `▶ Écoutez la piste ci-dessous et téléchargez-la en WAV. Dites-moi si vous voulez une variation : plus rapide, plus sombre, autre style, autre durée…`,
   473→            pending: false,
   474→            artifacts: [artifact],
   475→          })
   476→        } catch (e) {
   477→          updateMessage(sessionId, assistantId, {
   478→            content: '',
   479→            pending: false,
   480→            error: `La génération musicale a échoué : ${e instanceof Error ? e.message : 'erreur inconnue'}`,
   481→          })
   482→        }
   483→      }
